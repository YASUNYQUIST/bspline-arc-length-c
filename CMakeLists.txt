cmake_minimum_required(VERSION 3.16)
project(bspline_arc_length VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(bspline src/bspline.c)
target_include_directories(bspline PUBLIC include)
if(NOT MSVC)
  target_compile_options(bspline PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(bspline PUBLIC m)
endif()
add_executable(bspline_demo examples/demo.c)
target_link_libraries(bspline_demo PRIVATE bspline)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_bspline tests/test_bspline.c)
  target_link_libraries(test_bspline PRIVATE bspline)
  add_test(NAME core COMMAND test_bspline)
endif()
option(BSPLINE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(BSPLINE_SANITIZE AND NOT MSVC)
  target_compile_options(bspline PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bspline PUBLIC -fsanitize=address,undefined)
endif()
